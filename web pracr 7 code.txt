    <div class="bg-white w-full max-w-md p-6 rounded-2xl shadow-lg">

        <h1 class="text-3xl font-bold text-center text-blue-600 mb-6">
            Personal To-Do List
        </h1>

        <!-- Input Section -->
        <div class="flex gap-2 mb-5">
            <input
                type="text"
                id="taskInput"
                placeholder="Enter your task..."
                class="flex-1 border border-gray-300 rounded-lg px-4 py-2 
                       focus:outline-none focus:ring-2 focus:ring-blue-400"
            >

            <button
                onclick="addTask()"
                class="bg-blue-500 text-white px-4 py-2 rounded-lg 
                       hover:bg-blue-600">
                Add
            </button>
        </div>

        <!-- Task List -->
        <ul id="taskList" class="space-y-3"></ul>

    </div>

    <script>
        function addTask() {

            let input = document.getElementById("taskInput");
            let task = input.value.trim();

            if (task === "") {
                alert("Please enter a task!");
                return;
            }

            let li = document.createElement("li");

            li.className =
                "flex items-center justify-between bg-gray-100 p-3 rounded-lg";

            li.innerHTML = `
                <span
                    class="task-text cursor-pointer flex-1"
                    onclick="completeTask(this)">
                    ${task}
                </span>

                <button
                    onclick="deleteTask(this)"
                    class="bg-red-500 text-white px-3 py-1 rounded-lg 
                           hover:bg-red-600">
                    Delete
                </button>
            `;

            document.getElementById("taskList").appendChild(li);

            input.value = "";
        }

        function completeTask(element) {

            element.classList.toggle("line-through");
            element.classList.toggle("text-gray-400");
        }

        function deleteTask(button) {

            button.parentElement.remove();
        }
    </script>
