    <script>

        function calculateGrade() {

            let name = document.getElementById("name").value.trim();

            let java = Number(document.getElementById("java").value);
            let dsa = Number(document.getElementById("dsa").value);
            let web = Number(document.getElementById("web").value);

            // Validation
            if (name === "" ||
                java < 0 || java > 100 ||
                dsa < 0 || dsa > 100 ||
                web < 0 || web > 100) {

                alert("Please enter valid details and marks between 0 and 100.");
                return;
            }

            // Calculate total and percentage
            let total = java + dsa + web;
            let percentage = (total / 300) * 100;

            // Calculate grade
            let grade;

            if (percentage >= 90) {
                grade = "A+";
            }
            else if (percentage >= 80) {
                grade = "A";
            }
            else if (percentage >= 70) {
                grade = "B";
            }
            else if (percentage >= 60) {
                grade = "C";
            }
            else if (percentage >= 50) {
                grade = "D";
            }
            else {
                grade = "F";
            }

            // Display result
            document.getElementById("studentResult").innerHTML =
                "<strong>Student Name:</strong> " + name + "<br>" +
                "<strong>Total Marks:</strong> " + total + " / 300<br>" +
                "<strong>Percentage:</strong> " +
                percentage.toFixed(2) + "%<br>" +
                "<strong>Grade:</strong> " + grade;

            document.getElementById("result")
                .classList.remove("hidden");
        }

    </script>
