    <script>
        function calculateGrade() {

            let name = document.getElementById("name").value;

            let java = Number(document.getElementById("java").value);
            let python = Number(document.getElementById("python").value);
            let dsa = Number(document.getElementById("dsa").value);
            let web = Number(document.getElementById("web").value);

            // Calculate total
            let total = java + python + dsa + web;

            // Calculate percentage
            let percentage = (total / 400) * 100;

            // Calculate grade
            let grade;

            if (percentage >= 90) {
                grade = "A+";
            }
            else if (percentage >= 80) {
                grade = "A";
            }
            else if (percentage >= 70) {
                grade = "B";
            }
            else if (percentage >= 60) {
                grade = "C";
            }
            else if (percentage >= 50) {
                grade = "D";
            }
            else {
                grade = "F";
            }

            // Display result
            document.getElementById("studentName").innerText = name;
            document.getElementById("total").innerText = total;
            document.getElementById("percentage").innerText =
                percentage.toFixed(2);
            document.getElementById("grade").innerText = grade;

            document.getElementById("result").style.display = "block";
        }
    </script>
